cmake_minimum_required(VERSION 3.20)
project(clqr_reference_benchmark LANGUAGES C CXX ASM)

# Optional experiment harness, separate from the library's Bazel dependencies.
# Reference implementations are fetched by the runner, never vendored here.
set(CLQR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
# GTSAM uses native architecture tuning; apply the same setting to CLQR and
# the reference adapters instead of comparing native and generic C++ builds.
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:-march=native>")
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static comparison dependencies")
set(GEN_RICCATI_SOURCE_DIR "" CACHE PATH "Vanroye reference checkout")
set(BLASFEO_SOURCE_DIR "" CACHE PATH "BLASFEO checkout")
set(FACTOR_GRAPH_SOURCE_DIR "" CACHE PATH "Yang et al. reference checkout")
set(LAINE_SOURCE_DIR "" CACHE PATH "Unmodified Forrest Laine parallel_lqr checkout")
set(EIGEN_SOURCE_DIR "" CACHE PATH "Eigen headers for the optional Laine comparison")
option(CLQR_COMPARE_LAINE_REIMPLEMENTATION "Include our independent Laine reimplementation" OFF)

if(BLASFEO_SOURCE_DIR)
  set(BLASFEO_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(BLASFEO_TESTING OFF CACHE BOOL "" FORCE)
  set(BLAS_API ON CACHE BOOL "" FORCE)
  add_subdirectory("${BLASFEO_SOURCE_DIR}" blasfeo EXCLUDE_FROM_ALL)
endif()

add_library(clqr_paper_cpu STATIC "${CLQR_ROOT}/src/clqr.cc" "${CLQR_ROOT}/src/linalg.cc")
target_include_directories(clqr_paper_cpu PUBLIC "${CLQR_ROOT}/include")
add_executable(clqr_paper_fixture ../paper_fixture.cc)
target_include_directories(clqr_paper_fixture PRIVATE "${CLQR_ROOT}")
target_link_libraries(clqr_paper_fixture PRIVATE clqr_paper_cpu)
add_executable(clqr_cpu_benchmark ../paper_benchmark.cc)
target_include_directories(clqr_cpu_benchmark PRIVATE "${CLQR_ROOT}")
target_link_libraries(clqr_cpu_benchmark PRIVATE clqr_paper_cpu)
add_executable(clqr_reference_benchmark ../paper_benchmark.cc)
target_include_directories(clqr_reference_benchmark PRIVATE "${CLQR_ROOT}")
target_link_libraries(clqr_reference_benchmark PRIVATE clqr_paper_cpu)
if(GEN_RICCATI_SOURCE_DIR)
  if(NOT BLASFEO_SOURCE_DIR)
    message(FATAL_ERROR "GEN_RICCATI_SOURCE_DIR needs BLASFEO_SOURCE_DIR")
  endif()
  add_library(gen_riccati_reference STATIC
    "${GEN_RICCATI_SOURCE_DIR}/src/gen_riccati/gen_riccati.cpp"
    "${GEN_RICCATI_SOURCE_DIR}/src/gen_riccati/blasfeo_wrapper.cpp")
  target_include_directories(gen_riccati_reference PUBLIC
    "${GEN_RICCATI_SOURCE_DIR}/src/gen_riccati")
  target_link_libraries(gen_riccati_reference PUBLIC blasfeo)
  target_link_libraries(clqr_reference_benchmark PRIVATE gen_riccati_reference)
  target_compile_definitions(clqr_reference_benchmark PRIVATE CLQR_BENCHMARK_GEN_RICCATI)
  add_executable(clqr_adversarial_benchmark ../paper_benchmark.cc)
  target_include_directories(clqr_adversarial_benchmark PRIVATE "${CLQR_ROOT}")
  target_compile_definitions(clqr_adversarial_benchmark PRIVATE
    CLQR_BENCHMARK_ADVERSARIAL CLQR_BENCHMARK_GEN_RICCATI)
  target_link_libraries(clqr_adversarial_benchmark PRIVATE clqr_paper_cpu gen_riccati_reference)
endif()

if(FACTOR_GRAPH_SOURCE_DIR OR EIGEN_SOURCE_DIR)
  enable_testing()
  add_executable(quadratic_factor_test quadratic_factor_test.cc)
  target_include_directories(quadratic_factor_test PRIVATE "${CLQR_ROOT}")
  if(EIGEN_SOURCE_DIR)
    target_include_directories(quadratic_factor_test SYSTEM PRIVATE "${EIGEN_SOURCE_DIR}")
  endif()
  add_test(NAME quadratic_factor_test COMMAND quadratic_factor_test)
  add_executable(stationarity_audit_test stationarity_audit_test.cc)
  target_include_directories(stationarity_audit_test PRIVATE "${CLQR_ROOT}")
  target_link_libraries(stationarity_audit_test PRIVATE clqr_paper_cpu)
  if(EIGEN_SOURCE_DIR)
    target_include_directories(stationarity_audit_test SYSTEM PRIVATE "${EIGEN_SOURCE_DIR}")
  else()
    find_package(GTSAM CONFIG REQUIRED)
    target_link_libraries(stationarity_audit_test PRIVATE gtsam)
  endif()
  add_test(NAME stationarity_audit_test COMMAND stationarity_audit_test)
endif()

if(FACTOR_GRAPH_SOURCE_DIR)
  find_package(GTSAM CONFIG REQUIRED)
  target_include_directories(quadratic_factor_test PRIVATE "${FACTOR_GRAPH_SOURCE_DIR}/src")
  target_compile_definitions(quadratic_factor_test PRIVATE CLQR_BENCHMARK_GTSAM)
  target_link_libraries(quadratic_factor_test PRIVATE gtsam)
  target_link_libraries(clqr_reference_benchmark PRIVATE gtsam)
  target_include_directories(clqr_reference_benchmark PRIVATE "${FACTOR_GRAPH_SOURCE_DIR}/src")
  target_compile_definitions(clqr_reference_benchmark PRIVATE CLQR_BENCHMARK_GTSAM)
  add_executable(clqr_adversarial_factor_graph_benchmark ../paper_benchmark.cc)
  target_include_directories(clqr_adversarial_factor_graph_benchmark PRIVATE
    "${CLQR_ROOT}" "${FACTOR_GRAPH_SOURCE_DIR}/src")
  target_compile_definitions(clqr_adversarial_factor_graph_benchmark PRIVATE
    CLQR_BENCHMARK_ADVERSARIAL CLQR_BENCHMARK_GTSAM)
  target_link_libraries(clqr_adversarial_factor_graph_benchmark PRIVATE clqr_paper_cpu gtsam)
endif()

if(LAINE_SOURCE_DIR)
  if(NOT EXISTS "${EIGEN_SOURCE_DIR}/Eigen/Core")
    message(FATAL_ERROR "LAINE_SOURCE_DIR requires EIGEN_SOURCE_DIR")
  endif()
  file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/laine-include")
  file(CREATE_LINK "${EIGEN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/laine-include/eigen3" SYMBOLIC)
  add_library(laine_author_original STATIC
    "${LAINE_SOURCE_DIR}/src/trajectory.cpp"
    "${LAINE_SOURCE_DIR}/src/dynamics.cpp"
    "${LAINE_SOURCE_DIR}/src/running_constraint.cpp"
    "${LAINE_SOURCE_DIR}/src/endpoint_constraint.cpp"
    "${LAINE_SOURCE_DIR}/src/running_cost.cpp"
    "${LAINE_SOURCE_DIR}/src/terminal_cost.cpp"
    "${LAINE_SOURCE_DIR}/src/numerical_gradient.cpp")
  target_include_directories(laine_author_original PUBLIC "${LAINE_SOURCE_DIR}/src")
  target_include_directories(laine_author_original SYSTEM PUBLIC "${EIGEN_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/laine-include")
  target_compile_options(laine_author_original PRIVATE -include functional)
  enable_testing()
  add_executable(laine_author_adapter_test laine_author_adapter_test.cc)
  target_include_directories(laine_author_adapter_test PRIVATE "${CLQR_ROOT}")
  target_link_libraries(laine_author_adapter_test PRIVATE clqr_paper_cpu laine_author_original)
  add_test(NAME laine_author_adapter_test COMMAND laine_author_adapter_test)
  add_executable(clqr_laine_benchmark ../paper_benchmark.cc)
  target_include_directories(clqr_laine_benchmark PRIVATE "${CLQR_ROOT}")
  target_compile_definitions(clqr_laine_benchmark PRIVATE CLQR_BENCHMARK_LAINE)
  target_link_libraries(clqr_laine_benchmark PRIVATE clqr_paper_cpu laine_author_original)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME laine_author_benchmark_test
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/laine_author_benchmark_test.py"
            "$<TARGET_FILE:clqr_laine_benchmark>")
  add_executable(clqr_adversarial_laine_benchmark ../paper_benchmark.cc)
  target_include_directories(clqr_adversarial_laine_benchmark PRIVATE "${CLQR_ROOT}")
  target_compile_definitions(clqr_adversarial_laine_benchmark PRIVATE
    CLQR_BENCHMARK_ADVERSARIAL CLQR_BENCHMARK_LAINE)
  target_link_libraries(clqr_adversarial_laine_benchmark PRIVATE clqr_paper_cpu laine_author_original)
endif()

if(CLQR_COMPARE_LAINE_REIMPLEMENTATION)
  if(NOT EXISTS "${EIGEN_SOURCE_DIR}/Eigen/Core")
    message(FATAL_ERROR "CLQR_COMPARE_LAINE_REIMPLEMENTATION needs EIGEN_SOURCE_DIR")
  endif()
  add_library(laine_reimplementation STATIC "${CLQR_ROOT}/external_algorithms/laine_tomlin/solver.cc")
  target_include_directories(laine_reimplementation PUBLIC "${CLQR_ROOT}" "${CLQR_ROOT}/include")
  target_include_directories(laine_reimplementation SYSTEM PUBLIC "${EIGEN_SOURCE_DIR}")
  enable_testing()
  foreach(test_name dense_test solver_test fixture_test)
    add_executable(laine_${test_name} "${CLQR_ROOT}/external_algorithms/laine_tomlin/${test_name}.cc")
    target_include_directories(laine_${test_name} PRIVATE "${CLQR_ROOT}")
    target_link_libraries(laine_${test_name} PRIVATE laine_reimplementation)
    if(test_name STREQUAL "fixture_test")
      target_link_libraries(laine_${test_name} PRIVATE clqr_paper_cpu)
    endif()
    add_test(NAME laine_${test_name} COMMAND laine_${test_name})
  endforeach()
  foreach(target clqr_laine_reimplementation_benchmark clqr_adversarial_laine_reimplementation_benchmark)
    add_executable(${target} ../paper_benchmark.cc)
    target_include_directories(${target} PRIVATE "${CLQR_ROOT}")
    target_compile_definitions(${target} PRIVATE CLQR_BENCHMARK_LAINE_REIMPLEMENTATION)
    target_link_libraries(${target} PRIVATE clqr_paper_cpu laine_reimplementation)
  endforeach()
  target_compile_definitions(clqr_adversarial_laine_reimplementation_benchmark PRIVATE
    CLQR_BENCHMARK_ADVERSARIAL CLQR_BENCHMARK_INDEPENDENT_FIXTURES)
  foreach(target clqr_adversarial_benchmark clqr_adversarial_factor_graph_benchmark clqr_adversarial_laine_benchmark)
    if(TARGET ${target})
      target_include_directories(${target} SYSTEM PRIVATE "${EIGEN_SOURCE_DIR}")
      target_compile_definitions(${target} PRIVATE CLQR_BENCHMARK_INDEPENDENT_FIXTURES)
    endif()
  endforeach()
endif()
